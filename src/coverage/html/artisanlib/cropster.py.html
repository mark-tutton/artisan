            try:

                    res['extratimex'] = []
                if 'extratemp1' not in res:
                    res['extratemp1'] = []
                if 'extratemp2' not in res:
                    res['extratemp2'] = []
                if 'extramathexpression1' not in res:
                    res['extramathexpression1'] = []
                if 'extramathexpression2' not in res:
                    res['extramathexpression2'] = []
